action.enabled,
                containerColor = containerColor,
                contentColor = contentColor,
                height = segmentHeight,
                iconSize = segmentIconSize,
                isFirst = index == 0,
                isLast = index == actions.lastIndex,
                useWeight = false,
                contentDescription = action.contentDescription,
                content = action.content
            )
        }
    }
}
